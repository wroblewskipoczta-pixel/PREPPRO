Połącz z Google Sheets (5 minut)

Aby raporty z telefonów zawodników spływały na żywo do Twojego kalendarza, wklej adres URL skryptu Google Apps Script poniżej:

Kod do wklejenia w Arkuszu Google (Rozszerzenia → Apps Script)

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var d = JSON.parse(e.postData.contents);
  sheet.appendRow([new Date(), d.date, d.team, d.player, d.count, JSON.stringify(d.done)]);
  return ContentService.createTextOutput(JSON.stringify({ok:true}))
    .setMimeType(ContentService.MimeType.JSON);
}
function doGet() {
  var rows = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet().getDataRange().getValues();
  return ContentService.createTextOutput(JSON.stringify(rows))
    .setMimeType(ContentService.MimeType.JSON);
}
